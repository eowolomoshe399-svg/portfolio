# converts decimal to hex and 16-bit binary
def convert(n):
    hex_str = f"0x{n:04X}"
    bin16 = bin(n)[2:].zfill(16)
    if n < 32768: signed = n
    else: signed = n - 65536  # two's complement
    return hex_str, bin16, signed

# checks 65535 reads as -1 in signed 16-bit
